cmake_minimum_required(VERSION 3.16)
project(AttuneAnalysis LANGUAGES CXX)
add_library(attune_analysis STATIC src/BPMDetector.cpp src/HarmonicTempo.cpp src/KeyDetector.cpp)
target_compile_features(attune_analysis PUBLIC cxx_std_17)
target_include_directories(attune_analysis PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    add_executable(attune_demo demo.cpp)
    target_link_libraries(attune_demo PRIVATE attune_analysis)
    enable_testing()
    add_test(NAME generated_loop COMMAND attune_demo)
endif()
